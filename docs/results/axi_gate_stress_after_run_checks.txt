import axi.experiments.gate_stress as G
seg = int(sys.argv[1])
big = []
orig = G.Runtime.propose
def wrapped(self, text, *a, **k):
    if len(text) > G.BIG_TEXT:
        fb = len(self.pager._free); tb = len(self.pager._table)
        try:
            o = orig(self, text, *a, **k); res = (o.committed, o.stage, o.reason[:60] if o.reason else "")
        except Exception as e:
            o = None; res = ("raised", type(e).__name__, str(e)[:60])
        big.append(dict(n=len(text), res=res, free_before=fb, free_after=len(self.pager._free), table_before=tb, table_after=len(self.pager._table)))
        if o is None: raise
        return o
    return orig(self, text, *a, **k)

import sys
sys.path.insert(0, "/home/claude/axi-runtime/python")
import axi.experiments.gate_stress as G
from collections import Counter
seg = int(sys.argv[1])
orig = G.Runtime.propose
stats = Counter(); first = []
def wrapped(self, text, *a, **k):
    fb = list(self.pager._free); tb = dict(self.pager._table)
    o = orig(self, text, *a, **k)
    if not o.committed:
        stats["refused", o.stage] += 1
        fa = list(self.pager._free)
        if fa != fb:
            stats["refused_free_list_changed", o.stage] += 1
            if sorted(fa) == sorted(fb): stats["  of which same set, other order", o.stage] += 1
            if len(first) < 2: first.append((o.stage, len(text), (o.reason or "")[:80], len(fb), len(fa)))
        if self.pager._table != tb: stats["refused_table_changed", o.stage] += 1
    return o
G.Runtime.propose = wrapped
S = G.run_segment((seg, 20000))
print("seg", seg, "(first 20,000 proposals only)", dict(stats), first, "DONE", flush=True)
